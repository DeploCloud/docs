Deplo detects Next.js automatically from `next` in your `package.json` and
configures the build preset for you, no `deplo.json` needed for a standard
app.

<Tabs items={['bun', 'npm', 'pnpm']}>
<Tab value="bun">

```bash
bun install
deplo up
```

</Tab>
<Tab value="npm">

```bash
npm install
deplo up
```

</Tab>
<Tab value="pnpm">

```bash
pnpm install
deplo up
```

</Tab>
</Tabs>

## Supported rendering modes

<TypeTable
  type={{
    "Static (SSG)": {
      description: "Pages exported at build time, served from the edge cache",
      type: "supported",
    },
    "Server-side rendering": {
      description: "Runs per-request in Deplo's Node runtime",
      type: "supported",
    },
    "Route handlers": {
      description: "app/api and app/**/route.ts, run in the Node runtime",
      type: "supported",
    },
    "Edge runtime": {
      description: "export const runtime = 'edge' functions",
      type: "supported",
    },
    "Incremental Static Regeneration": {
      description: "on-demand and time-based revalidation",
      type: "supported",
    },
  }}
/>

<Callout type="info">
  Image Optimization works out of the box, `next/image` requests are served
  through Deplo's built-in image proxy.
</Callout>
